# PhotoTTS - iOS APP 详细设计文档

## 1. 项目概述

### 1.1 项目简介
PhotoTTS 是一个基于iOS的智能拍照文字转语音阅读应用，专为7-12岁学生设计。用户可以通过拍照或选择相册图片，将图片中的文字直接转换为语音，提供生动有趣的阅读体验。

### 1.2 核心特性
- 支持拍照和相册选择图片
- 调用豆包大模型OCR API进行文字识别
- 调用火山引擎TTS API进行文字转语音
- 儿童友好的音频播放控制
- 支持多语言（中文、英文等）
- 简洁的代码结构，易于维护

### 1.3 技术规格
- 开发语言: Swift 5.0+
- 最低支持版本: iOS 14.0+
- 架构模式: MVVM + Coordinator
- UI框架: SwiftUI + UIKit混合使用
- 设计风格: 儿童友好、简洁现代

### 3.3 网络服务模块 (Network Service Module)

#### 3.3.1 核心类设计

NetworkService.swift
```swift
class NetworkService: ObservableObject {
    private let session = URLSession.shared
    private let ocrService: OCRService
    private let ttsService: TTSService
    
    init() {
        self.ocrService = OCRService()
        self.ttsService = TTSService()
    }
    
    // 文字识别
    func recognizeText(from imageData: Data, completion: @escaping (Result<String, Error>) -> Void) {
        ocrService.recognizeText(from: imageData, completion: completion)
    }
    
    // 文字转语音
    func synthesizeSpeech(from text: String, completion: @escaping (Result<Data, Error>) -> Void) {
        ttsService.synthesizeSpeech(from: text, completion: completion)
    }
    
    // 组合功能：图片转语音
    func convertImageToSpeech(from image: UIImage, completion: @escaping (Result<Data, Error>) -> Void) {
        // 先进行OCR识别
        guard let imageData = image.jpegData(compressionQuality: 0.8) else {
            completion(.failure(NetworkError.imageProcessingFailed))
            return
        }
        
        recognizeText(from: imageData) { [weak self] result in
            switch result {
            case .success(let text):
                // 再进行TTS合成
                self?.synthesizeSpeech(from: text, completion: completion)
            case .failure(let error):
                completion(.failure(error))
            }
        }
    }
}

enum NetworkError: Error, LocalizedError {
    case imageProcessingFailed
    case networkError(Error)
    case apiError(String)
    case invalidResponse
    
    var errorDescription: String? {
        switch self {
        case .imageProcessingFailed:
            return "图片处理失败"
        case .networkError(let error):
            return "网络错误: \(error.localizedDescription)"
        case .apiError(let message):
            return "API错误: \(message)"
        case .invalidResponse:
            return "响应数据无效"
        }
    }
}
```

class SettingsManager: ObservableObject {
    static let shared = SettingsManager()
    
    // API设置
    @Published var apiKey: String? {
        didSet {
            UserDefaults.standard.set(apiKey, forKey: "api_key")
        }
    }
    
    // 语音设置
    @Published var voiceSettings: VoiceSettings = VoiceSettings.default {
        didSet {
            if let data = try? JSONEncoder().encode(voiceSettings) {
                UserDefaults.standard.set(data, forKey: "voice_settings")
            }
        }
    }
    
    // 语言设置
    @Published var supportedLanguages: [String] = ["zh", "en"] {
        didSet {
            UserDefaults.standard.set(supportedLanguages, forKey: "supported_languages")
        }
    }
    
    private init() {
        loadSettings()
    }
    
    private func loadSettings() {
        apiKey = UserDefaults.standard.string(forKey: "api_key")
        
        if let data = UserDefaults.standard.data(forKey: "voice_settings"),
           let settings = try? JSONDecoder().decode(VoiceSettings.self, from: data) {
            voiceSettings = settings
        }
        
        if let languages = UserDefaults.standard.stringArray(forKey: "supported_languages") {
            supportedLanguages = languages
        }
    }
}

## 5. 数据流设计

### 5.1 图片处理流程

```
单张：用户拍照 → 图像预处理 → 本地存储 → OCR API → 文字识别 → TTS API → 语音合成 → 本地缓存 → 播放音频
批量：用户拍摄或选择图片 → 图片预处理 → 本地存储 → 图片队列管理 → OCR API(批量) → 文字识别和拼接 → TTS API → 语音合成 → 本地缓存 → 播放音频
```

相比单张图片处理，批量图片处理 新增了 图片队列管理功能。批量图片处理，详细数据流程如下：

class BatchProcessingState: ObservableObject {
    // 图片管理
    @Published var selectedImages: [UIImage] = []
    @Published var currentImageIndex: Int = 0
    
    // 处理状态
    @Published var isBatchProcessing: Bool = false
    @Published var batchProgress: Float = 0.0
    @Published var currentOperation: String = ""
    
    // 结果数据
    @Published var combinedText: String = ""
    @Published var batchAudioData: Data?
    @Published var isPlaying: Bool = false
    
    // 耗时统计
    @Published var ocrDuration: TimeInterval = 0.0
    @Published var ttsDuration: TimeInterval = 0.0
    
    // 图片收集模式
    @Published var collectionMode: CollectionMode = .single
    @Published var maxImages: Int = 10
    
    enum CollectionMode {
        case single      // 单张模式
        case batch       // 批量模式
    }
    
    // 添加图片到队列
    func addImage(_ image: UIImage) {
        guard selectedImages.count < maxImages else { return }
        selectedImages.append(image)
    }
    
    // 从队列中删除图片
    func removeImage(at index: Int) {
        guard index < selectedImages.count else { return }
        selectedImages.remove(at: index)
        if currentImageIndex >= selectedImages.count {
            currentImageIndex = max(0, selectedImages.count - 1)
        }
    }
    
    // 清空所有图片
    func clearAllImages() {
        selectedImages.removeAll()
        currentImageIndex = 0
        combinedText = ""
        batchAudioData = nil
    }
    
    // 开始批量处理
    func startBatchProcessing() {
        guard !selectedImages.isEmpty else { return }
        isBatchProcessing = true
        batchProgress = 0.0
        currentOperation = "开始批量处理..."
    }
    
    // 完成批量处理
    func completeBatchProcessing(text: String, audioData: Data?) {
        isBatchProcessing = false
        batchProgress = 1.0
        currentOperation = "处理完成"
        combinedText = text
        batchAudioData = audioData
    }
}
